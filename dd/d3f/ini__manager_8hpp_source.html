344 auto header(std::string_view text, std::size_t column) -> result<void>
345 {
346 const auto end = text.find(']');
347 if (end == std::string_view::npos)
348 {
349 return std::unexpected(diagnostic(error_reason::invalid_header, column));
350 }
351 auto tail = text.substr(end + 1);
352 if (!tail.empty() && tail.front() != ' ' && tail.front() != '\t')
353 {
354 return std::unexpected(
355 diagnostic(error_reason::invalid_header, column + end + 1));
356 }
357 tail = trim(tail);
358 if (!tail.empty() && !comment(tail.front()))
359 {
360 return std::unexpected(
361 diagnostic(error_reason::invalid_header, column + end + 1));
362 }
363 const auto name = trim(text.substr(1, end - 1));
364 if (!valid_section(name))
365 {
366 return std::unexpected(diagnostic(error_reason::invalid_section, column + 1));
367 }
368 if (!m_data.contains(name) && m_data.size() >= m_options.max_sections)
369 {
370 return std::unexpected(diagnostic(error_reason::resource_limit, column));
371 }
372 m_current = name;
373 m_data.try_emplace(m_current);
374 return {};
375 }
376
377 auto store(std::string_view name, std::string_view value, std::size_t column)
378 -> result<void>
379 {
380 auto group = m_data.find(m_current);
381 if (group == m_data.end())
382 {
383 if (m_data.size() >= m_options.max_sections)
384 {
385 return std::unexpected(diagnostic(error_reason::resource_limit, column));
386 }
387 group = m_data.try_emplace(m_current).first;
388 }
389 const bool duplicate = group->second.contains(name);
390 if (duplicate && m_options.duplicates == duplicate_policy::reject)
391 {
392 auto failure = diagnostic(error_reason::duplicate_key, column);
393 failure.section_name = m_current;
394 failure.key_name = name;
395 return std::unexpected(std::move(failure));
396 }
397 if (!duplicate && m_keys >= m_options.max_keys)
398 {
399 return std::unexpected(diagnostic(error_reason::resource_limit, column));
400 }
401 group->second.insert_or_assign(name, std::string(value));
402 if (!duplicate)
403 {
404 ++m_keys;
405 }
406 return {};
407 }
408
409 auto entry(std::string_view text, std::size_t column) -> result<void>
410 {
411 const auto separator = text.find_first_of(m_options.allow_colon ? "=:" : "=");
412 if (separator == std::string_view::npos)
413 {
414 return std::unexpected(diagnostic(error_reason::missing_separator, column));
415 }
416 const auto name = trim(text.substr(0, separator));
417 auto value = text.substr(separator + 1);
418 if (m_options.inline_comments)
419 {
420 value = value.substr(0, inline_comment(value));
421 }
422 value = trim(value);
423 if (auto validated = validate_entry({m_current}, {name}, value, m_options);
424 !validated)
425 {
426 auto failure = std::move(validated.error());
427 failure.op = operation::parse;
428 failure.line = m_line_number;
429 failure.column = column;
430 return std::unexpected(std::move(failure));
431 }
432 return store(name, value, column);
433 }

447 {
448 constexpr std::string_view bom = "\xEF\xBB\xBF";
449 std::size_t offset = 0;
450 if (m_line_number == 1 && raw.starts_with(bom))
451 {
452 raw.remove_prefix(bom.size());

456 {

462 }

465 {
466 return {};
467 }
468 const auto column =
469 static_cast<std::size_t>(text.data() - raw.data()) + offset + 1;
470 return text.front() == '[' ? header(text, column) : entry(text, column);
471 }
472 void next_line()
473 {
474 ++m_line_number;
475 }
476 auto finish() && -> data_map
477 {
478 return std::move(m_data);
479 }
480 auto read_failure(error failure, std::size_t column) const -> error
481 {
482 failure.line = m_line_number;
483 failure.column = column;
484 return failure;
485 }

489class document_reader
490{
491 document_parser m_parser;
492 parse_options m_options;
493 std::string m_line;
494 std::size_t m_total = 0;
495 bool m_carriage_return = false;
496
497 public:
498 explicit document_reader(parse_options options)
499 : m_parser(options), m_options(options)
500 {
501 }
502 auto consume(char byte) -> result<void>
503 {
504 if (m_total >= m_options.max_input_bytes)
505 {
506 return std::unexpected(
507 m_parser.diagnostic(error_reason::resource_limit, m_line.size() + 1));
508 }
509 ++m_total;
510 if (m_carriage_return && byte != '\n')
511 {
512 return std::unexpected(
513 m_parser.diagnostic(error_reason::invalid_control, m_line.size() + 1));
514 }
515 if (byte == '\n')
516 {
517 if (auto processed = m_parser.process(m_line); !processed)
518 {
519 return processed;
520 }
521 m_line.clear();
522 m_carriage_return = false;
523 m_parser.next_line();
524 }
525 else if (byte == '\r')
526 {
527 m_carriage_return = true;
528 }
529 else
530 {
531 if (m_line.size() >= m_options.max_line_bytes)
532 {
533 return std::unexpected(
534 m_parser.diagnostic(error_reason::resource_limit, m_line.size() + 1));
535 }
536 m_line.push_back(byte);
537 }
538 return {};
539 }
540 auto finish() && -> result<data_map>
541 {
542 if (m_carriage_return)
543 {
544 return std::unexpected(
545 m_parser.diagnostic(error_reason::invalid_control, m_line.size() + 1));
546 }
547 if (!m_line.empty())
548 {
549 if (auto processed = m_parser.process(m_line); !processed)
550 {
551 return std::unexpected(std::move(processed.error()));
552 }
553 }
554 return std::move(m_parser).finish();
555 }
556 auto read_failure(error failure) const -> error
557 {
558 return m_parser.read_failure(std::move(failure), m_line.size() + 1);
559 }

562// Copy the small source callable; it is invoked repeatedly and never consumed.
563template <class Next>
564auto parse(Next next, const parse_options &options) -> result<data_map>
565{
566 document_reader reader(options);
567 for (;;)
568 {
569 auto byte = next();
570 if (!byte)
571 {
572 return std::unexpected(reader.read_failure(std::move(byte.error())));
573 }
574 if (!*byte)
575 {
576 return std::move(reader).finish();
577 }

579 {
580 return std::unexpected(std::move(consumed.error()));
581 }

674 -> result<std::string>
675{
676 std::string output;
677 const auto limit_error = [] -> std::unexpected<ini::error> {
678 return std::unexpected(

680 };

687 if (section_name.size() > options.max_input_bytes - output.size())

693 };

698 return std::unexpected(

704 return limit_error();

707 {
708 return limit_error();
709 }
710 if (entries.size() > options.max_keys - keys)
711 {
712 return limit_error();

714 keys += entries.size();

716 !written)

719 }
720 if (!append("\n"))

790 count = ::read(m_handle, buffer, size);
791 }
792 if (count < 0)

794 return std::unexpected(io_error(operation::read, native_error()));
795 }
796#endif

799 auto write(std::string_view bytes) const -> result<void>
800 {
801 while (!bytes.empty())
802 {
803 const auto amount = (std::min)(bytes.size(), 1024UZ * 1024);
804#ifdef _WIN32
805 DWORD count = 0;
806 if (!WriteFile(m_handle, bytes.data(), static_cast<DWORD>(amount), &count,
807 nullptr))
808 return std::unexpected(io_error(operation::write, native_error()));
809#else
810 auto count = ::write(m_handle, bytes.data(), amount);
811 while (count < 0 && errno == EINTR)
812 {
813 count = ::write(m_handle, bytes.data(), amount);
814 }
815 if (count < 0)
816 {
817 return std::unexpected(io_error(operation::write, native_error()));
818 }
819#endif
820 if (count == 0)
821 {
822 return std::unexpected(io_error(operation::write));
823 }
824 bytes.remove_prefix(static_cast<std::size_t>(count));
825 }
826 return {};
827 }

829 {
830#ifndef _WIN32

832 0)
833 {
834 return std::unexpected(io_error(operation::permissions, native_error()));
835 }
836#else
837 (void)mode;
838#endif
839 return {};
840 }

842 {
843#ifdef _WIN32

845 return {};

848#else

851 return {};
852 }
853 const auto handle = std::exchange(m_handle, -1);
854 // Never retry close(EINTR): the descriptor may already have been released.
855 const bool failed = ::close(handle) != 0;
856#endif

859 return std::unexpected(io_error(operation::close, native_error()));

861 return {};

1028// Internal injectable operation sequence, also used by deterministic failure tests.
1029template <class Ops> auto atomic_write(Ops &ops, std::string_view bytes) -> result<void>
1030{
1031 if (auto outcome = ops.inspect(); !outcome)
1032 {
1033 return outcome;
1034 }
1035 bool owned = false;
1036 struct cleanup_guard
1037 {
1038 Ops *ops;
1039 bool *owned;
1040 cleanup_guard(Ops &operations, bool &active) : ops(&operations), owned(&active)
1041 {
1042 }
1043 cleanup_guard(const cleanup_guard &) = delete;
1044 cleanup_guard(cleanup_guard &&) = delete;
1045 auto operator=(const cleanup_guard &) -> cleanup_guard & = delete;
1046 auto operator=(cleanup_guard &&) -> cleanup_guard & = delete;
1047 ~cleanup_guard()
1048 {
1049 if (*owned)
1050 {
1051 (void)ops->cleanup();
1052 }
1053 }
1054 };
1055 const cleanup_guard guard{ops, owned};
1056 constexpr unsigned max_attempts = 128;
1057 for (unsigned attempt = 0; attempt < max_attempts; ++attempt)
1058 {
1059 auto outcome = ops.create();
1060 if (!outcome)
1061 {
1062 return std::unexpected(std::move(outcome.error()));
1063 }
1064 if (*outcome)
1065 {
1066 owned = true;
1067 break;
1068 }
1069 }
1070 if (!owned)
1071 {
1072 return std::unexpected(io_error(operation::create_temp,
1073 std::make_error_code(std::errc::file_exists)));
1074 }
1075 const auto fail = [&](error diagnostic) -> result<void> {
1076 diagnostic.cleanup_code = ops.cleanup();
1077 owned = false;
1078 if (diagnostic.cleanup_code)
1079 {
1080 diagnostic.temporary_path = ops.temporary;
1081 }
1082 return std::unexpected(std::move(diagnostic));
1083 };
1084 if (auto outcome = ops.write(bytes); !outcome)
1085 {
1086 return fail(std::move(outcome.error()));
1087 }
1088 if (auto outcome = ops.permissions(); !outcome)
1089 {
1090 return fail(std::move(outcome.error()));
1091 }
1092 if (auto outcome = ops.close(); !outcome)
1093 {
1094 return fail(std::move(outcome.error()));
1095 }
1096 if (auto outcome = ops.replace(); !outcome)
1097 {
1098 return fail(std::move(outcome.error()));
1099 }
1100 owned = false;
1101 return {};
1102}
