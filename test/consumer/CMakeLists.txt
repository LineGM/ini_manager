cmake_minimum_required(VERSION 4.0...4.4)
project(consumer LANGUAGES CXX)
set(original_flags "${CMAKE_CXX_FLAGS}")
if(MODE STREQUAL "installed")
    find_package(ini_manager 1.0 REQUIRED)
elseif(MODE STREQUAL "fetch")
    include(FetchContent)
    FetchContent_Declare(ini_manager SOURCE_DIR "${INI_SOURCE}")
    FetchContent_MakeAvailable(ini_manager)
else()
    # Even an explicitly enabled developer option must not affect consumers.
    set(ini_manager_DEVELOPER_MODE ON CACHE BOOL "")
    add_subdirectory("${INI_SOURCE}" ini-manager)
endif()
if(TARGET ini_manager_test OR TARGET run-examples OR TARGET format-check OR TARGET ini_manager_fuzz OR TARGET ini_manager_benchmark OR TARGET tidy-check OR TARGET ini_manager_ini_manager_verify_interface_header_sets)
    message(FATAL_ERROR "Developer targets leaked into consumer")
endif()
if(NOT "${original_flags}" STREQUAL "${CMAKE_CXX_FLAGS}")
    message(FATAL_ERROR "Consumer compiler flags changed")
endif()
foreach(property IN ITEMS INTERFACE_COMPILE_OPTIONS INTERFACE_LINK_OPTIONS)
    get_target_property(developer_flags ini_manager::ini_manager ${property})
    if(developer_flags)
        message(FATAL_ERROR "Developer options leaked through ${property}: ${developer_flags}")
    endif()
endforeach()
add_executable(consumer "${INI_SOURCE}/test/source/header_b.cpp" "${INI_SOURCE}/test/source/header_a.cpp")
target_link_libraries(consumer PRIVATE ini_manager::ini_manager)
enable_testing()
add_test(NAME consumer_runs COMMAND consumer)
